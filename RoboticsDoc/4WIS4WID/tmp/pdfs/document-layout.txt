  代码块​
  1   #include <math.h>​
  2   #include <stddef.h>​
  3   /**​
  4   * @brief     将世界坐标系平动速度转换为机体坐标系平动速度。​
  5   *​
  6   坐标约定：​
  7   世界系和机体系均为右手坐标系；​
  8   yaw_rad > 0 表示机体相对世界坐标系逆时针旋转；​
  9   vx_world_mps、vy_world_mps 为世界坐标系期望速度；​
 10   vx_body_mps、vy_body_mps 为四舵轮逆运动学所需的机体系速度。​
 11   */​
 12   static void app_chassis_world_to_body_velocity(​
 13        float vx_world_mps,​
 14        float vy_world_mps,​
 15        float yaw_rad,​
 16        float *vx_body_mps,​
 17        float *vy_body_mps)​
 18   {​
 19        float cos_yaw;​
 20        float sin_yaw;​
 21   ​
 22        if ((vx_body_mps == NULL)​
 23               || (vy_body_mps == NULL)) {​
 24               return;​
 25        }​
 26   ​
 27        cos_yaw = cosf(yaw_rad);​
 28        sin_yaw = sinf(yaw_rad);​
 29   ​
 30        /*​
 31          * [Vx_body]     [ cos(yaw)   sin(yaw)] [Vx_world]​
 32          * [Vy_body] = [-sin(yaw)     cos(yaw)] [Vy_world]​
 33          *​
 34          *   即：​
 35          * V_body = R(-yaw) * V_world​
 36          */​
 37        *vx_body_mps =​
 38               vx_world_mps * cos_yaw​
    39              + vy_world_mps * sin_yaw;​
    40   ​
    41        *vy_body_mps =​
    42              -vx_world_mps * sin_yaw​
    43              + vy_world_mps * cos_yaw;​
    44   }​
